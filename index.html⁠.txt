<script>
  let currentQ = 1;
  let audioCtx = null;
  let bgmInterval = null;
  let loopVoiceInterval = null;

  function initAudio() {
    if (!audioCtx) {
      audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }
  }

  // 背景壓抑音樂
  function startCreepyBGM() {
    initAudio();
    if (bgmInterval) return;

    const droneOsc = audioCtx.createOscillator();
    const droneGain = audioCtx.createGain();
    droneOsc.type = 'sawtooth';
    droneOsc.frequency.setValueAtTime(45, audioCtx.currentTime);

    const filter = audioCtx.createBiquadFilter();
    filter.type = 'lowpass';
    filter.frequency.setValueAtTime(180, audioCtx.currentTime);

    droneGain.gain.setValueAtTime(0.08, audioCtx.currentTime);

    droneOsc.connect(filter);
    filter.connect(droneGain);
    droneGain.connect(audioCtx.destination);
    droneOsc.start();

    bgmInterval = setInterval(() => {
      if (Math.random() < 0.6) {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'sine';
        
        const freqs = [587.33, 622.25, 880.00, 932.33, 1174.66];
        const freq = freqs[Math.floor(Math.random() * freqs.length)];
        
        osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.01, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 2.5);

        osc.connect(gain);
        gain.connect(audioCtx.destination);

        osc.start();
        osc.stop(audioCtx.currentTime + 2.5);
      }
    }, 1500);
  }

  function nextQuestion() {
    initAudio();
    const currentCard = document.querySelector(`.question-card[data-q="${currentQ}"]`);
    if (currentCard) currentCard.classList.remove('active');

    currentQ++;
    const nextCard = document.querySelector(`.question-card[data-q="${currentQ}"]`);
    if (nextCard) {
      nextCard.classList.add('active');
    }

    if (currentQ === 14) {
      playYandereLaughter();
    }
  }

  function startAnalysis() {
    initAudio();
    const q7 = document.querySelector(`.question-card[data-q="7"]`);
    q7.classList.remove('active');

    const loadingScreen = document.getElementById('loading-screen');
    loadingScreen.classList.add('active');

    const progressBar = document.getElementById('progress-bar');
    const progressText = document.getElementById('loading-percent');

    const stages = [
      { pct: 12, delay: 400 },
      { pct: 27, delay: 900 },
      { pct: 53, delay: 1500 },
      { pct: 86, delay: 2100 },
      { pct: 100, delay: 2800 }
    ];

    stages.forEach((stage) => {
      setTimeout(() => {
        progressBar.style.width = stage.pct + '%';
        progressText.innerText = stage.pct + '%';
        playGlitchSound();

        if (stage.pct === 100) {
          setTimeout(() => {
            document.body.classList.add('dark-mode');
            loadingScreen.classList.remove('active');
            startCreepyBGM();
            playBeepSound();

            currentQ = 8;
            const q8 = document.querySelector(`.question-card[data-q="8"]`);
            q8.classList.add('active');
          }, 800);
        }
      }, stage.delay);
    });
  }

  function playGlitchSound() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'square';
    osc.frequency.setValueAtTime(300 + Math.random() * 600, audioCtx.currentTime);
    gain.gain.setValueAtTime(0.03, audioCtx.currentTime);
    gain.gain.linearRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);

    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + 0.1);
  }

  function playBeepSound() {
    if (!audioCtx) return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();

    osc.type = 'sawtooth';
    osc.frequency.setValueAtTime(150, audioCtx.currentTime);
    gain.gain.setValueAtTime(0.1, audioCtx.currentTime);

    osc.connect(gain);
    gain.connect(audioCtx.destination);

    osc.start();
    osc.stop(audioCtx.currentTime + 0.6);
  }

  function triggerJumpscareAndNext() {
    initAudio();
    playScreamSound();

    const overlay = document.getElementById('jumpscare-overlay');
    overlay.classList.add('active');

    setTimeout(() => {
      overlay.classList.remove('active');
      nextQuestion();
    }, 600);
  }

  function playScreamSound() {
    if (!audioCtx) return;
    const bufferSize = audioCtx.sampleRate * 0.6;
    const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
    const output = buffer.getChannelData(0);

    for (let i = 0; i < bufferSize; i++) {
      output[i] = Math.random() * 2 - 1;
    }

    const whiteNoise = audioCtx.createBufferSource();
    whiteNoise.buffer = buffer;

    const osc = audioCtx.createOscillator();
    osc.type = 'sawtooth';
    osc.frequency.setValueAtTime(800, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(120, audioCtx.currentTime + 0.6);

    const gain = audioCtx.createGain();
    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.6);

    whiteNoise.connect(gain);
    osc.connect(gain);
    gain.connect(audioCtx.destination);

    whiteNoise.start();
    osc.start();
    whiteNoise.stop(audioCtx.currentTime + 0.6);
    osc.stop(audioCtx.currentTime + 0.6);
  }

  function rejectAndRedirect() {
    window.location.href = "https://youtu.be/XqZsoesa55w?si=08yulVTkUdHdzFSf&autoplay=1";
  }

  function playYandereLaughter() {
    if ('speechSynthesis' in window) {
      const laughUtterance = new SpeechSynthesisUtterance('嘻嘻，嘻嘻嘻，哈哈哈哈！');
      laughUtterance.lang = 'zh-TW';
      laughUtterance.pitch = 1.5;
      laughUtterance.rate = 1.1;
      window.speechSynthesis.speak(laughUtterance);
    }
  }

  function goToQuestion15() {
    nextQuestion();
    speakYuanYi();

    if (!loopVoiceInterval) {
      loopVoiceInterval = setInterval(() => {
        speakYuanYi();
      }, 1800);
    }
  }

  function speakYuanYi() {
    if ('speechSynthesis' in window) {
      const msg = new SpeechSynthesisUtterance('願意');
      msg.lang = 'zh-TW';
      msg.pitch = 1.2;
      msg.rate = 0.8;
      window.speechSynthesis.speak(msg);
    }
  }

  function finishTest() {
    if (loopVoiceInterval) {
      clearInterval(loopVoiceInterval);
      loopVoiceInterval = null;
    }

    const q15 = document.querySelector(`.question-card[data-q="15"]`);
    q15.classList.remove('active');

    showFinalResult();
  }

  function showFinalResult() {
    const resultCard = document.getElementById('result-card');
    resultCard.classList.add('active');

    const lines = document.querySelectorAll('.result-text');
    let delay = 300;

    lines.forEach((line, index) => {
      setTimeout(() => {
        line.classList.add('show');

        if (index === lines.length - 1) {
          setTimeout(() => {
            triggerFlash();
          }, 1200);
        }
      }, delay);

      delay += 1200;
    });
  }

  function triggerFlash() {
    const flash = document.getElementById('flash-overlay');
    flash.classList.add('flash');

    setTimeout(() => {
      flash.style.backgroundColor = "#ffffff";
    }, 50);
  }
